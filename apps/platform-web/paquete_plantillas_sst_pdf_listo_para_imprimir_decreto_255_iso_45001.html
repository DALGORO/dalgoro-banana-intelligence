1) Criterio para definir el riesgo dominante

Definición: nivel (Bajo/Medio/Alto) derivado del IPER considerando severidad, probabilidad, población/tiempo de exposición, límites de exposición y presencia de trabajos especiales o riesgos de seguridad industriales mayores.

  • ALTO (automático): existen trabajos especiales, riesgos con potencial de accidente mayor o exposiciones por sobre límites permitidos.
  • MEDIO: exposiciones próximas a límites o requieren controles de ingeniería relevantes.
  • BAJO: exposiciones ampliamente por debajo de límites, gestionables con controles administrativos y EPP estándar.

Áreas mixtas: prevalece el nivel más alto que afecte ≥ 30% de horas-hombre o involucre peligros críticos; si no se cumple, adoptar el nivel modal y justificar en el IPER.

4-bis.2 Generador automático (Google Apps Script – mínimo viable)

Preparación: crea en Drive una carpeta SST_OUTPUT y una hoja SST_MAESTRO con pestañas: Centros (RazonSocial, RUC, Centro, Clasificacion, Riesgo, FA_PPRL, FA_RHS, FA_ORG, Responsable, etc.), Plantillas (Nombre, Tipo, DocID/SheetID), Documentos (salida). Usa los mismos {{campos}} que aparecen en las plantillas.

/** Google Apps Script – Generador de documentos SST (Docs→PDF) */
function generateDocs() {
  const ss = SpreadsheetApp.getActive();
  const centros = ss.getSheetByName('Centros').getDataRange().getValues();
  const plantillas = ss.getSheetByName('Plantillas').getDataRange().getValues();
  const out = ss.getSheetByName('Documentos');
  const folder = DriveApp.getFoldersByName('SST_OUTPUT').hasNext() ?
                 DriveApp.getFoldersByName('SST_OUTPUT').next() : DriveApp.createFolder('SST_OUTPUT');
  const headerC = centros.shift();
  const mapPlant = Object.fromEntries(plantillas.slice(1).map(r => [r[0], r[2]])); // Nombre → DocID

  out.clear();
  out.appendRow(['Centro','Plantilla','Doc ID','Doc URL','PDF URL','Fecha']);

  centros.slice(1).forEach(row => {
    if (!row.length || !row[0]) return;
    const data = Object.fromEntries(headerC.map((k,i)=>[k, row[i]]));

    // Ejemplos de combinaciones a generar por clasificación
    const bundle = buildBundleByClass(data.Clasificacion);

    bundle.forEach(nombreTpl => {
      const tplId = mapPlant[nombreTpl];
      if (!tplId) return;
      const copy = DriveApp.getFileById(tplId).makeCopy(`${data.Centro} - ${nombreTpl} (${new Date().toISOString().slice(0,10)})`, folder);
      const doc = DocumentApp.openById(copy.getId());
      const body = doc.getBody();
      replacePlaceholders(body, data);
      doc.saveAndClose();
      const pdf = DriveApp.getFileById(copy.getId()).getAs('application/pdf');
      const pdfFile = folder.createFile(pdf).setName(copy.getName()+'.pdf');
      out.appendRow([data.Centro, nombreTpl, copy.getId(), copy.getUrl(), pdfFile.getUrl(), new Date()]);
    });
  });
}

function replacePlaceholders(body, data){
  Object.keys(data).forEach(k => { body.replaceText(`{{${k}}}`, String(data[k]||'')); });
}

function buildBundleByClass(cl){
  // Ajusta nombres según fila "Plantillas"
  if (cl === 'Micro') return ['Nombramiento Responsable','PPRL – Modelo','Plan Anual Capacitaciones','Plan de Emergencias','Bitácora SUT'];
  if (cl === 'Pequeña') return ['Nombramiento Responsable','RHS – Modelo','Acta Elección Delegado','Acta Posesión Delegado','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];
  if (cl === 'Mediana') return ['Nombramiento Responsable','RHS – Modelo','Acta Elección Comité','Reglamento Interno Comité','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];
  if (cl === 'Grande') return ['Nombramiento Responsable','RHS – Modelo','Acta Elección Comité','Reglamento Interno Comité','Servicio de Salud – Evidencias','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];
  return ['Nombramiento Responsable'];
}

Personaliza la lista buildBundleByClass y los nombres exactos de la pestaña Plantillas. Para reemplazos avanzados (tablas, listas), podemos ampliar replacePlaceholders y añadir mapeos por Riesgo Dominante.

/** === Utilidades de clasificación y decisiones === */
function classifyByWorkers(n) {
  n = Number(n||0);
  if (n <= 9) return 'Micro';
  if (n <= 49) return 'Pequeña';
  if (n <= 99) return 'Mediana';
  return 'Grande';
}
function decideResponsible(clasif, riesgo) {
  const r = String(riesgo||'').toLowerCase();
  return (r === 'alto') ? 'Técnico' : 'Monitor';
}
function decideOrg(clasif) {
  if (clasif === 'Pequeña') return 'Delegado';
  if (clasif === 'Mediana' || clasif === 'Grande') return 'Comité';
  return '';
}

  if (cl === 'Pequeña') pack = ['Nombramiento Responsable','RHS – Modelo','Acta Elección Delegado','Acta Posesión Delegado','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];
  if (cl === 'Mediana') pack = ['Nombramiento Responsable','RHS – Modelo','Acta Elección Comité','Reglamento Interno Comité','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];
  if (cl === 'Grande') pack = ['Nombramiento Responsable','RHS – Modelo','Acta Elección Comité','Reglamento Interno Comité','Servicio de Salud – Evidencias','Plan Anual Capacitaciones','Programa Psicosocial','Programa ATD','Bitácora SUT'];

  // Riesgo Alto añade permisos críticos
  if (String(riesgo).toLowerCase() === 'alto') pack = pack.concat(permisosCriticos);

  // Adiciones por actividad
  return pack.concat(activityAddons(act));
}

/** === Carpetas de salida: Clasificación/Actividad/Riesgo/Centro === */
function ensureFolders(rootName, clasif, actividad, riesgo, centro) {
  const root = DriveApp.getFoldersByName(rootName).hasNext()
    ? DriveApp.getFoldersByName(rootName).next()
    : DriveApp.createFolder(rootName);
  const f1 = getOrCreate(root, clasif);
  const f2 = getOrCreate(f1, actividad);
  const f3 = getOrCreate(f2, riesgo);
  return getOrCreate(f3, centro);
}
function getOrCreate(parent, name){
  const it = parent.getFoldersByName(name);
  return it.hasNext() ? it.next() : parent.createFolder(name);
}

/** === Generación principal === */
function generateDocsV2() {
  const ss = SpreadsheetApp.getActive();
  const centrosSh = ss.getSheetByName('Centros');
  const plantsSh  = ss.getSheetByName('Plantillas');
  const out       = ss.getSheetByName('Documentos');
  const centros = centrosSh.getDataRange().getValues();
  const plants  = plantsSh.getDataRange().getValues();
  const headC = centros.shift();
  const headP = plants.shift();
  const idx = Object.fromEntries(headC.map((k,i)=>[k,i]));
  const mapPlant = Object.fromEntries(plants.map(r => [r[0], r[2]])); // Nombre → DocID

  out.clear();
  out.appendRow(['Centro','Clasificación','Actividad','Riesgo','Plantilla','Doc ID','Doc URL','PDF URL','Observación','Fecha']);

  centros.forEach(row => {
    if (!row || !row.length || !row[idx['Centro']]) return;
    const data = Object.fromEntries(headC.map((k,i)=>[k, row[i]]));

    // Clasificación automática por Nº trabajadores
    const nTrab = Number(data['NTrabajadores']||0);
    const clasifCalc = classifyByWorkers(nTrab);
    const clasifManual = data['Clasificacion'];
    let clasif = clasifManual || clasifCalc;
    let obs = '';
    if (clasifManual && clasifManual !== clasifCalc) {
      obs = `Clasificación ajustada de ${clasifManual} a ${clasifCalc} por Nº trabajadores`;
      clasif = clasifCalc;
    }

    const riesgo = String(data['Riesgo']||'Bajo');
    const actividad = String(data['Actividad']||'Otros');
    data['Clasificacion'] = clasif;
    data['ResponsableSST'] = decideResponsible(clasif, riesgo);
    data['OrganismoParitario'] = decideOrg(clasif);

    const folder = ensureFolders('SST_OUTPUT', clasif, actividad, riesgo, data['Centro']);
    const bundle = buildBundleByClassRiskActivity(clasif, riesgo, actividad);
    const dateTag = new Date().toISOString().slice(0,10);

    bundle.forEach(nombreTpl => {
      const tplId = mapPlant[nombreTpl];
      if (!tplId) {
        out.appendRow([data['Centro'], clasif, actividad, riesgo, nombreTpl, '', '', '', 'Plantilla no encontrada en Plantillas', new Date()]);
        return;
      }
      const baseName = `${data['Centro']} - ${clasif} - ${actividad} - ${riesgo} - ${nombreTpl} - ${dateTag}`;
      const copy = DriveApp.getFileById(tplId).makeCopy(baseName, folder);
      const doc  = DocumentApp.openById(copy.getId());
      const body = doc.getBody();

      Object.keys(data).forEach(k => { body.replaceText(`{{${k}}}`, String(data[k]||'')); });
      doc.saveAndClose();

      const pdfBlob = DriveApp.getFileById(copy.getId()).getAs('application/pdf');
      const pdfFile = folder.createFile(pdfBlob).setName(baseName + '.pdf');

      out.appendRow([data['Centro'], clasif, actividad, riesgo, nombreTpl, copy.getId(), copy.getUrl(), pdfFile.getUrl(), obs, new Date()]);
    });
  });
}

Usa generateDocsV2() para segmentación completa. Asegúrate que la pestaña Plantillas contenga los nombres exactos listados en los bundles y que la pestaña Centros tenga la columna Actividad con una de las seis opciones.
